JOIN farm_areas a ON a.name_key = v.area_key

WHERE NOT EXISTS (SELECT 1 FROM flocks f WHERE f.cycle_id = c.id AND f.name = v.name);

-- Biến động "Nhập đàn" cho mỗi đàn mẫu (dùng để tính tổng nhập, khối lượng vào và FCR)
INSERT INTO flock_movements (flock_id, type, movement_date, quantity, weight_kg, unit_price, related_flock_id, note, created_by, created_at)
SELECT f.id, 'IMPORT', f.import_date, f.import_quantity, f.import_quantity * f.import_avg_weight, NULL, NULL, 'Dữ liệu mẫu', 'seed', f.created_at
FROM flocks f
WHERE NOT EXISTS (SELECT 1 FROM flock_movements m WHERE m.flock_id = f.id AND m.type = 'IMPORT');

-- Số con của chuồng = tổng tồn các đàn trong lứa đang nuôi; chuồng có lứa đang nuôi ở trạng thái ACTIVE
UPDATE barns b SET current_quantity = COALESCE((SELECT SUM(f.quantity) FROM flocks f JOIN farming_cycles c ON c.id = f.cycle_id
                                                WHERE c.barn_id = b.id AND c.status = 'ACTIVE'), 0);
UPDATE barns b SET status = 'ACTIVE'
WHERE EXISTS (SELECT 1 FROM farming_cycles c WHERE c.barn_id = b.id AND c.status = 'ACTIVE');

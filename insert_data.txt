UPDATE barns b SET current_quantity = COALESCE((SELECT SUM(f.quantity) FROM flocks f JOIN farming_cycles c ON c.id = f.cycle_id
                                                WHERE c.barn_id = b.id AND c.status = 'ACTIVE'), 0);
-- Trạng thái chuồng theo lứa nuôi: có lứa đang nuôi là ACTIVE, còn lại INACTIVE (kể cả giá trị CLOSED cũ đã bỏ).
UPDATE barns b SET status = CASE WHEN EXISTS (SELECT 1 FROM farming_cycles c WHERE c.barn_id = b.id AND c.status = 'ACTIVE')
                                 THEN 'ACTIVE' ELSE 'INACTIVE' END;

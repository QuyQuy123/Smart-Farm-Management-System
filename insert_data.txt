-- Tạo cặp kho riêng (FEED + MEDICINE) cho mỗi khu nuôi nếu khu chưa có
INSERT INTO warehouse_categories (name, name_key, description, area_id, kind, created_at, updated_at)
SELECT k.prefix || regexp_replace(fa.name, '^[Kk]hu\s+', ''),
       LOWER(k.prefix || regexp_replace(fa.name, '^[Kk]hu\s+', '')),
       k.note || fa.name,
       fa.id,
       k.kind,
       TIMESTAMP '2026-10-01 08:00:00',
       TIMESTAMP '2026-10-01 08:00:00'
FROM farm_areas fa
CROSS JOIN (VALUES
  ('Kho cám ', 'FEED', 'Cám cho các chuồng '),
  ('Kho thuốc ', 'MEDICINE', 'Thuốc thú y và vắc-xin cho ')
) AS k(prefix, kind, note)
WHERE NOT EXISTS (
  SELECT 1
  FROM warehouse_categories w
  LEFT JOIN farm_area_warehouses faw ON faw.warehouse_id = w.id
  WHERE (w.area_id = fa.id OR faw.area_id = fa.id) AND w.kind = k.kind
)
ON CONFLICT (name_key) DO UPDATE SET
  area_id = COALESCE(warehouse_categories.area_id, EXCLUDED.area_id),
  kind = COALESCE(warehouse_categories.kind, EXCLUDED.kind);

  AND NOT EXISTS (SELECT 1 FROM product_categories p WHERE p.warehouse_id = w.id)
  AND NOT EXISTS (SELECT 1 FROM purchase_invoice_items p WHERE p.warehouse_id = w.id)
  AND NOT EXISTS (SELECT 1 FROM internal_stock_vouchers s WHERE s.source_warehouse_id = w.id OR s.target_warehouse_id = w.id)
  AND NOT EXISTS (SELECT 1 FROM warehouse_workers ww WHERE ww.warehouse_id = w.id);

DECLARE
  v_tables TEXT;
BEGIN
  SELECT string_agg(format('%I', tablename), ', ')
  INTO v_tables
  FROM pg_tables
  WHERE schemaname = 'public'

      'work_task_checks', 'work_tasks', 'barn_workers', 'farm_area_workers', 'warehouse_workers',

      'measurement_units', 'product_types', 'customer_contacts', 'farms',

-- Gắn purchase_invoice_item_id đại diện cho các lô tồn kho mới nhất (không trùng UNIQUE)
UPDATE inventory_items i
SET purchase_invoice_item_id = picked.pii_id,
    lot_number = picked.lot_number,
    expiry_date = picked.expiry_date
FROM (
  SELECT DISTINCT ON (pii.warehouse_id, pii.product_category_id)
         pii.id AS pii_id,
         pii.warehouse_id,
         pii.product_category_id,
         pii.lot_number,
         pii.expiry_date
  FROM purchase_invoice_items pii
  JOIN purchase_invoices pi ON pi.id = pii.invoice_id
  WHERE pi.stock_status = 'Đã nhập kho'
    AND pii.warehouse_id IS NOT NULL
    AND pii.product_category_id IS NOT NULL
  ORDER BY pii.warehouse_id, pii.product_category_id, pi.invoice_date DESC, pii.id DESC
) picked
WHERE i.warehouse_id = picked.warehouse_id
  AND i.product_id = picked.product_category_id
  AND i.purchase_invoice_item_id IS NULL;

  FROM flock_movements m WHERE m.flock_id = f.id

      WHEN GREATEST(0, COALESCE((

      ), f.import_quantity)) = 0 THEN 'ARCHIVED'

    END;

-- Sinh daily_task_logs + feed_records + daily_stock_allocations cho 7 ngày (06/11 – 12/11/2026)
-- theo đúng task_key & đơn vị quy đổi của DailyPlanService
DO $$
DECLARE
  r_day       RECORD;
  r_step      RECORD;
  v_age       INT;
  v_heads     INT;
  v_planned   NUMERIC(14,3);
  v_per_meal  NUMERIC(14,3);
  v_packs     NUMERIC(19,6);
  v_cost      NUMERIC(16,2);
  v_unit_px   NUMERIC(19,2);
  v_area_item BIGINT;
  v_pack_kg   NUMERIC(14,3);
  v_area_px   NUMERIC(19,2);
  v_feed_id   BIGINT;
  v_log_id    BIGINT;
  v_meal      INT;
  v_task_key  TEXT;
  v_actor     TEXT;
BEGIN
  FOR r_day IN
    SELECT d::date AS work_date, c.id AS cycle_id, c.start_date, b.id AS barn_id, b.area_id, b.name AS barn_name
    FROM generate_series(DATE '2026-11-06', DATE '2026-11-12', INTERVAL '1 day') d
    CROSS JOIN farming_cycles c
    JOIN barns b ON b.id = c.barn_id
    WHERE c.status = 'ACTIVE' AND d::date >= c.start_date
  LOOP
    SELECT COALESCE(MAX(a.email), 'congnhan@smartfarm.com')
    INTO v_actor
    FROM barn_workers bw JOIN accounts a ON a.id = bw.account_id
    WHERE bw.barn_id = r_day.barn_id;

    SELECT COALESCE(SUM(
      f.import_quantity - COALESCE((
        SELECT SUM(CASE WHEN m.type IN ('DEATH','CULL','HARVEST','TRANSFER_OUT') THEN m.quantity ELSE -m.quantity END)
        FROM flock_movements m
        WHERE m.flock_id = f.id AND m.type <> 'IMPORT' AND m.movement_date <= r_day.work_date
      ), 0)
    ), 0)
    INTO v_heads
    FROM flocks f WHERE f.cycle_id = r_day.cycle_id;

    v_age := (r_day.work_date - r_day.start_date) + 1;

    FOR r_step IN
      SELECT DISTINCT ON (s.category, s.inventory_item_id, LOWER(TRIM(s.name)))
             s.id AS step_id, s.category, s.name, s.quantity, s.meals_per_day, s.inventory_item_id,
             tmpl_ii.product_id AS tmpl_product_id
      FROM farming_program_steps s
      JOIN flocks f ON f.program_id = s.program_id
      LEFT JOIN inventory_items tmpl_ii ON tmpl_ii.id = s.inventory_item_id
      WHERE f.cycle_id = r_day.cycle_id
        AND v_age BETWEEN s.day_from AND s.day_to
        AND s.category IN ('FEED', 'TEMPERATURE')
      ORDER BY s.category, s.inventory_item_id, LOWER(TRIM(s.name)), s.id
    LOOP
      IF r_step.category = 'TEMPERATURE' THEN
        v_task_key := 'TEMPERATURE:N' || LEFT(LOWER(REGEXP_REPLACE(TRIM(r_step.name), '\s+', ' ', 'g')), 90);
        INSERT INTO daily_task_logs (
          cycle_id, task_date, task_key, category, task_name, meal, planned_quantity, actual_quantity,
          usage_unit, inventory_item_id, stock_quantity, unit_price, cost, feed_record_id, expense_id,
          deviated, note, done_by, done_at
        )
        VALUES (
          r_day.cycle_id, r_day.work_date, v_task_key, 'TEMPERATURE', r_step.name, NULL,
          NULL, NULL, NULL, NULL, NULL, NULL, NULL, NULL, NULL,
          FALSE, 'Nhiệt độ chuồng ổn định theo ngưỡng quy trình', v_actor, r_day.work_date::timestamp + TIME '08:30:00'
        )
        ON CONFLICT (cycle_id, task_date, task_key) DO NOTHING;

      ELSIF r_step.category = 'FEED' AND r_step.inventory_item_id IS NOT NULL THEN
        SELECT ii.id, COALESCE(mu.content_quantity, 25.000), ii.price
        INTO v_area_item, v_pack_kg, v_area_px
        FROM inventory_items ii
        JOIN warehouse_categories w ON w.id = ii.warehouse_id
        LEFT JOIN measurement_units mu ON mu.id = ii.unit_id
        WHERE w.area_id = r_day.area_id AND w.kind = 'FEED' AND ii.product_id = r_step.tmpl_product_id
        ORDER BY ii.id ASC LIMIT 1;

        IF v_area_item IS NOT NULL THEN
          v_planned := ROUND((v_heads * r_step.quantity) / 1000.0, 3);
          v_per_meal := ROUND(v_planned / GREATEST(COALESCE(r_step.meals_per_day, 1), 1), 3);

          FOR v_meal IN 1..GREATEST(COALESCE(r_step.meals_per_day, 1), 1) LOOP
            -- Hôm nay (12/11): hoàn thành cữ 1 buổi sáng, để cữ 2/3/4 chờ công nhân bấm trên giao diện
            IF r_day.work_date < DATE '2026-11-12' OR v_meal = 1 THEN
              v_task_key := 'FEED:I' || r_step.inventory_item_id || ':M' || v_meal;
              v_packs := ROUND(v_per_meal / v_pack_kg, 6);
              v_cost := ROUND(v_packs * v_area_px, 0);
              v_unit_px := CASE WHEN v_packs > 0 THEN ROUND(v_cost / v_packs, 2) ELSE 0 END;

              INSERT INTO feed_records (cycle_id, record_date, feed_name, quantity_kg, unit_price, note, created_by, created_at)
              SELECT r_day.cycle_id, r_day.work_date, pc.name, v_per_meal,
                     ROUND(v_cost / NULLIF(v_per_meal, 0), 2),
                     'Việc hôm nay · cữ ' || v_meal || '/' || COALESCE(r_step.meals_per_day, 1),
                     v_actor, r_day.work_date::timestamp + (TIME '07:00:00' + ((v_meal - 1) * INTERVAL '4 hours'))
              FROM product_categories pc WHERE pc.id = r_step.tmpl_product_id
              RETURNING id INTO v_feed_id;

              INSERT INTO daily_task_logs (
                cycle_id, task_date, task_key, category, task_name, meal, planned_quantity, actual_quantity,
                usage_unit, inventory_item_id, stock_quantity, unit_price, cost, feed_record_id, expense_id,
                deviated, note, done_by, done_at
              )
              SELECT r_day.cycle_id, r_day.work_date, v_task_key, 'FEED', pc.name, v_meal,
                     v_per_meal, v_per_meal, 'KG', v_area_item, v_packs, v_unit_px, v_cost, v_feed_id, NULL,
                     FALSE, '', v_actor, r_day.work_date::timestamp + (TIME '07:15:00' + ((v_meal - 1) * INTERVAL '4 hours'))
              FROM product_categories pc WHERE pc.id = r_step.tmpl_product_id
              RETURNING id INTO v_log_id;

              INSERT INTO daily_stock_allocations (log_id, inventory_item_id, quantity)
              VALUES (v_log_id, v_area_item, v_packs);
            END IF;
          END LOOP;
        END IF;
      END IF;
    END LOOP;
  END LOOP;
END $$;

    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    name VARCHAR(150) NOT NULL UNIQUE,

    fund_type VARCHAR(30) NOT NULL DEFAULT 'BANK',

    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    code VARCHAR(30) NOT NULL UNIQUE,
    tx_date DATE NOT NULL,
    type VARCHAR(10) NOT NULL,

-- Tạo cặp kho riêng (FEED + MEDICINE) cho mỗi khu nuôi nếu khu chưa có
INSERT INTO warehouse_categories (name, name_key, description, area_id, kind, created_at, updated_at)
SELECT k.prefix || regexp_replace(fa.name, '^[Kk]hu\s+', ''),
       LOWER(k.prefix || regexp_replace(fa.name, '^[Kk]hu\s+', '')),
       k.note || fa.name,
       fa.id,
       k.kind,
       TIMESTAMP '2026-10-01 08:00:00',
       TIMESTAMP '2026-10-01 08:00:00'
FROM farm_areas fa
CROSS JOIN (VALUES
  ('Kho cám ', 'FEED', 'Cám cho các chuồng '),
  ('Kho thuốc ', 'MEDICINE', 'Thuốc thú y và vắc-xin cho ')
) AS k(prefix, kind, note)
WHERE NOT EXISTS (
  SELECT 1
  FROM warehouse_categories w
  LEFT JOIN farm_area_warehouses faw ON faw.warehouse_id = w.id
  WHERE (w.area_id = fa.id OR faw.area_id = fa.id) AND w.kind = k.kind
)
ON CONFLICT (name_key) DO UPDATE SET
  area_id = COALESCE(warehouse_categories.area_id, EXCLUDED.area_id),
  kind = COALESCE(warehouse_categories.kind, EXCLUDED.kind);

-- Đồng bộ liên kết Khu nuôi <-> Kho hàng vào bảng trung gian farm_area_warehouses
INSERT INTO farm_area_warehouses (area_id, warehouse_id)
SELECT area_id, id
FROM warehouse_categories
WHERE area_id IS NOT NULL
ON CONFLICT (area_id, warehouse_id) DO NOTHING;

-- Dọn kho chung cũ không còn hàng hóa / chứng từ / liên kết (an toàn cho DB nâng cấp)

WHERE w.area_id IS NULL

  AND NOT EXISTS (SELECT 1 FROM inventory_items i WHERE i.warehouse_id = w.id)
  AND NOT EXISTS (SELECT 1 FROM product_categories p WHERE p.warehouse_id = w.id)
  AND NOT EXISTS (SELECT 1 FROM purchase_invoice_items p WHERE p.warehouse_id = w.id)
  AND NOT EXISTS (SELECT 1 FROM internal_stock_vouchers s WHERE s.source_warehouse_id = w.id OR s.target_warehouse_id = w.id)
  AND NOT EXISTS (SELECT 1 FROM warehouse_workers ww WHERE ww.warehouse_id = w.id);

FROM barns b
JOIN farm_areas fa ON fa.id = b.area_id
JOIN farming_cycles c ON c.barn_id = b.id AND c.status = 'ACTIVE'
JOIN accounts a ON a.email = CASE fa.name_key
  WHEN 'khu mía thịt' THEN 'congnhan@smartfarm.com'
  WHEN 'khu j thịt' THEN 'hung@smartfarm.com'
  ELSE 'hang@smartfarm.com'
END
WHERE NOT EXISTS (SELECT 1 FROM barn_workers w WHERE w.barn_id = b.id)

INSERT INTO warehouse_workers (warehouse_id, account_id)
SELECT DISTINCT w.id, faw.account_id
FROM farm_area_workers faw
JOIN warehouse_categories w ON w.area_id = faw.area_id
  OR EXISTS (SELECT 1 FROM farm_area_warehouses aw WHERE aw.area_id = faw.area_id AND aw.warehouse_id = w.id)
ON CONFLICT (warehouse_id, account_id) DO NOTHING;
